5. Paper-style preprocessing¶

The paper describes

  • converting the raw Atari image to grayscale
  • downsampling it to 110×84
  • cropping it to 84×84.
  • stack four consecutive frames to obtain an 84×84×4 state representation.
import cv2
from collections import deque

def preprocess_frame(frame):
    gray = cv2.cvtColor(frame, cv2.COLOR_RGB2GRAY)
    resized = cv2.resize(gray, (84, 110), interpolation=cv2.INTER_AREA)
    top = (110 - 84) // 2
    cropped = resized[top:top+84, :]
    return cropped.astype(np.uint8)

processed = preprocess_frame(obs)

print("Processed frame shape:", processed.shape)

plt.axis("off")
plt.title("84×84 Preprocessed Frame")
plt.show()

Processed frame shape: (84, 84)

class FrameStack:
    def __init__(self, n_frames=4):
        self.n_frames = n_frames
        self.frames = deque(maxlen=n_frames)

    def reset(self, frame):
        p = preprocess_frame(frame)
        self.frames.clear()
        for _ in range(self.n_frames):
            self.frames.append(p)
        return np.stack(self.frames, axis=0)

    def append(self, frame):
        self.frames.append(preprocess_frame(frame))
        return np.stack(self.frames, axis=0)

stacker = FrameStack(4)
state = stacker.reset(obs)

print("Stacked state shape:", state.shape)

6. DQN network¶

The network below follows the core architecture reported in the 2013 paper:

  • Conv1: 16 filters, 8×8, stride 4, ReLU
  • Conv2: 32 filters, 4×4, stride 2, ReLU
  • Fully connected: 256 ReLU units
  • Output: one Q-value per valid action
class DQN(nn.Module):
    def __init__(self, in_channels, n_actions):
        super().__init__()

        self.features = nn.Sequential(
            nn.Conv2d(in_channels, 16, kernel_size=8, stride=4),
            nn.ReLU(),
            nn.Conv2d(16, 32, kernel_size=4, stride=2),
            nn.ReLU()
        )

        self.fc = nn.Sequential(
            nn.Flatten(),
            nn.Linear(32 * 9 * 9, 256),
            nn.ReLU()
        )

        self.head = nn.Linear(256, n_actions)

    def forward(self, x):
        x = x.float() / 255.0
        x = self.features(x)
        x = self.fc(x)
        return self.head(x)

model = DQN(4, env.action_space.n).to(DEVICE)

7. Experience Replay¶

Experience replay stores transitions and samples random minibatches for training.

from collections import deque

class ReplayBuffer:
    def __init__(self, capacity):
        self.buffer = deque(maxlen=capacity)

    def add(self, state, action, reward, next_state, done):
        self.buffer.append((state, action, reward, next_state, done))

    def sample(self, batch_size):
        batch = random.sample(self.buffer, batch_size)
        states, actions, rewards, next_states, dones = zip(*batch)
        return (
            np.stack(states),
            np.asarray(actions, dtype=np.int64),
            np.asarray(rewards, dtype=np.float32),
            np.stack(next_states),
            np.asarray(dones, dtype=np.float32)
        )

    def __len__(self):
        return len(self.buffer)

replay = ReplayBuffer(100_000)
print("Replay buffer initialized. Capacity:", 100_000)

8. DQN agent¶

The implementation uses:

  • RMSProp
  • minibatch size 32
  • discount factor γ = 0.99
  • reward clipping to −1, 0, +1
  • ε-greedy exploration
  • evaluation ε = 0.05

The training budget below is smaller than the original paper's experiment.

BATCH_SIZE = 32
GAMMA = 0.99
LR = 0.00025
TRAIN_FRAMES = 50_000
LEARNING_STARTS = 1_000
TRAIN_FREQUENCY = 4
REPLAY_CAPACITY = 100_000
EPS_START = 1.0
EPS_END = 0.1
EPS_DECAY = 100_000

class DQNAgent:
    def __init__(self, n_actions):
        self.device = torch.device(DEVICE)
        self.n_actions = n_actions
        self.online = DQN(4, n_actions).to(self.device)
        self.optimizer = torch.optim.RMSprop(
            self.online.parameters(),
            lr=LR,
            alpha=0.95,
            eps=0.01
        )
        self.replay = ReplayBuffer(REPLAY_CAPACITY)
        self.total_steps = 0

    def epsilon(self):
        fraction = min(1.0, self.total_steps / EPS_DECAY)
        return EPS_START + fraction * (EPS_END - EPS_START)

    def act(self, state, evaluate=False):
        eps = 0.05 if evaluate else self.epsilon()

        if random.random() < eps:
            return random.randrange(self.n_actions)

        x = torch.from_numpy(np.asarray(state)).unsqueeze(0).to(self.device)
        with torch.no_grad():
            return int(self.online(x).argmax(1).item())

    def update(self):
        if len(self.replay) < BATCH_SIZE:
            return None

        states, actions, rewards, next_states, dones = self.replay.sample(BATCH_SIZE)

        states = torch.from_numpy(states).to(self.device)
        actions = torch.from_numpy(actions).to(self.device)
        rewards = torch.from_numpy(rewards).to(self.device)
        next_states = torch.from_numpy(next_states).to(self.device)
        dones = torch.from_numpy(dones).to(self.device)

        current_q = self.online(states).gather(1, actions.unsqueeze(1)).squeeze(1)

        with torch.no_grad():
            next_q = self.online(next_states).max(1).values
            target = rewards + GAMMA * (1.0 - dones) * next_q

        loss = nn.functional.smooth_l1_loss(current_q, target)

        self.optimizer.zero_grad(set_to_none=True)
        loss.backward()
        torch.nn.utils.clip_grad_norm_(self.online.parameters(), 10.0)
        self.optimizer.step()

        return float(loss.item())

agent = DQNAgent(env.action_space.n)
print("Agent initialized on:", DEVICE)

9. Training¶

os.makedirs("/content/results", exist_ok=True)

training_log = "/content/results/training_log.csv"
checkpoint_path = "/content/results/checkpoint.pt"

with open(training_log, "w", newline="", encoding="utf-8") as f:
    writer = csv.writer(f)
    writer.writerow(["frame", "episode", "episode_reward", "epsilon", "loss"])

    frame = 0
    episode = 0

    while frame < TRAIN_FRAMES:
        obs, _ = env.reset(seed=SEED + episode)
        stacker = FrameStack(4)
        state = stacker.reset(obs)

        episode_reward = 0.0
        losses = []
        terminated = truncated = False

        while not (terminated or truncated) and frame < TRAIN_FRAMES:
            action = agent.act(state)

            next_obs, reward, terminated, truncated, _ = env.step(action)

            next_state = stacker.append(next_obs)
            clipped_reward = 1.0 if reward > 0 else (-1.0 if reward < 0 else 0.0)

            agent.replay.add(
                state,
                action,
                clipped_reward,
                next_state,
                terminated or truncated
            )

            state = next_state
            episode_reward += reward
            frame += 1
            agent.total_steps += 1

            if frame >= LEARNING_STARTS and frame % TRAIN_FREQUENCY == 0:
                loss = agent.update()
                if loss is not None:
                    losses.append(loss)

        mean_loss = float(np.mean(losses)) if losses else np.nan

        writer.writerow([
            frame,
            episode,
            episode_reward,
            agent.epsilon(),
            mean_loss
        ])
        f.flush()

        print(
            f"Episode {episode:4d} | "
            f"frame {frame:7d}/{TRAIN_FRAMES} | "
            f"reward {episode_reward:7.2f} | "
            f"epsilon {agent.epsilon():.3f}"
        )

        episode += 1

torch.save({
    "model": agent.online.state_dict(),
    "optimizer": agent.optimizer.state_dict(),
    "steps": agent.total_steps
}, checkpoint_path)

print("\nTraining complete.")

12. Evaluation¶

The paper evaluated using an ε-greedy policy with ε = 0.05.

The following cell performs a small evaluation run and writes the measured scores to a CSV.

# Load the trained model
checkpoint = torch.load(checkpoint_path, map_location=DEVICE)
agent.online.load_state_dict(checkpoint["model"])
agent.online.eval()

evaluation_rewards = []
EVAL_EPISODES = 5

for ep in range(EVAL_EPISODES):
    obs, _ = env.reset(seed=1000 + ep)
    stacker = FrameStack(4)
    state = stacker.reset(obs)

    total_reward = 0.0
    terminated = truncated = False

    while not (terminated or truncated):
        action = agent.act(state, evaluate=True)
        obs, reward, terminated, truncated, _ = env.step(action)
        state = stacker.append(obs)
        total_reward += reward

    evaluation_rewards.append(total_reward)
    print(f"Evaluation episode {ep+1}: reward = {total_reward:.2f}")

evaluation_csv = "/content/results/evaluation_results.csv"

with open(evaluation_csv, "w", newline="", encoding="utf-8") as f:
    writer = csv.writer(f)
    writer.writerow(["episode", "reward"])

    for i, reward in enumerate(evaluation_rewards, 1):
        writer.writerow([i, reward])

    writer.writerow([])
    writer.writerow(["mean_reward", np.mean(evaluation_rewards)])
    writer.writerow(["std_reward", np.std(evaluation_rewards)])

print("\nMean evaluation reward:", np.mean(evaluation_rewards))
print("Evaluation results:", evaluation_csv)
